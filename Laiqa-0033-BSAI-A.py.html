# Task 1
limit = {
    "Full Press": (40, 55),
    "Bettery Volt": (26, 32),
    "Cabin Temp": (18, 27),
    "signal": (60, 100)
}


def check_reading(sensor, value):
    """check if reading is normal or anomaly"""
    if limit[sensor][0] <= value <= limit[sensor][1]:
        return "NORMAL"
    else:
        return "ANOMALY"


def sensor_summary(values):
    """Find minimum, maximum and average."""
    mini = values[0]
    maxi = values[0]
    total = 0

    for v in values:
        if v < mini:
            mini = v
        if v > maxi:
            maxi = v
        total += v
    avg = total / len(values)
    return mini, maxi, avg


def subsystem_status(anomaly_count):
    """Return GO or NO-GO."""
    if anomaly_count >= 2:
        return "NO-GO"
    return "GO"


def save_report(filename, report_text):
    """Save report in a file."""
    file = open(filename, "w")
    file.write(report_text)
    file.close()


#task 2
Reading = {
    "Propulsion": {"Full Press": [47, 56, 52, 54]},
    "Power": {"Bettery Volt": [27, 27, 25, 28]},
    "Thermal": {"Cabin Temp": [22, 25, 28, 29]},
    "COMMS": {"signal": [91, 58, 74, 88]}
}

total = 0
anomalies = 0
launch = "GO"

for subsystem in Reading:
    count = 0

    for sensor in Reading[subsystem]:
        values = Reading[subsystem][sensor]

        for value in values:
            status = check_reading(sensor, value)
            print(sensor, value, status)
            total += 1

            if status == "ANOMALY":
                count += 1
                anomalies += 1

        print("Min, Max, Avg:", sensor_summary(values))

    print(subsystem, subsystem_status(count))

    if count >= 2:
        launch = "NO-GO"
        print("NO-GO subsystem:", subsystem)

print("Total readings:", total)
print("Total anomalies:", anomalies)
print("LAUNCH DECISION:", launch)

# for report save in file

name = input("Enter your name: ")
roll_no = input("Enter your roll number: ")

report = "NASA LAUNCH REPORT\n"
report += "Name: " + name + "\n"
report += "Roll Number: " + roll_no + "\n"
report += "Total Readings: " + str(total) + "\n"
report += "Total Anomalies: " + str(anomalies) + "\n"
report += "Launch Decision: " + launch + "\n"

save_report("mission_report.txt", report)

print("Report saved successfully!")
