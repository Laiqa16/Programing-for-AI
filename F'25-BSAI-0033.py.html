students = []

subject_names = [
    "Programming",
    "Mathematics",
    "Artificial Intelligence"
]

student_name = input("Enter student name (-1 to stop): ")

# Sentinel-controlled while loop
while student_name != "-1":

    total = 0
    marks = []

    # Process the three subjects using a for loop
    for subject in subject_names:
        mark = float(input("Enter marks in " + subject + ": "))

        marks.append(mark)
        total += mark

    average = total / 3

    # Pass condition: average must be at least 50
    if average >= 50:
        result = "Pass"
    else:
        result = "Fail"

    # Store the student's information
    students.append([student_name, marks, total, average, result])

    print("\nStudent Name:", student_name)
    print("Total Marks:", total)
    print("Average Marks:", round(average, 2))
    print("Result:", result)
    print("-" * 40)

    # Ask for the next student
    student_name = input("Enter student name (-1 to stop): ")


# Analyze the class after data entry
if len(students) == 0:
    print("\nNo student records were entered.")

else:
    student_count = 0
    class_total = 0
    highest_average = students[0][3]
    lowest_average = students[0][3]
    highest_student = students[0][0]
    lowest_student = students[0][0]
    pass_count = 0
    fail_count = 0

    # Calculate class statistics
    for student in students:
        student_count += 1
        class_total += student[3]

        if student[3] > highest_average:
            highest_average = student[3]
            highest_student = student[0]

        if student[3] < lowest_average:
            lowest_average = student[3]
            lowest_student = student[0]

        if student[4] == "Pass":
            pass_count += 1
        else:
            fail_count += 1

    class_average = class_total / student_count

    # Count students above the class average
    above_average_count = 0

    for student in students:
        if student[3] > class_average:
            above_average_count += 1

    # Display class statistics
    print("\n========== CLASS REPORT ==========")
    print("Number of Students:", student_count)
    print("Class Average:", round(class_average, 2))
    print("Highest Average:", round(highest_average, 2))
    print("Highest-Scoring Student:", highest_student)
    print("Lowest Average:", round(lowest_average, 2))
    print("Lowest-Scoring Student:", lowest_student)
    print("Passing Students:", pass_count)
    print("Failing Students:", fail_count)
    print("Above Class Average:", above_average_count)

    # Nested loops: compare every student with every other student
    print("\n======= STUDENT COMPARISONS =======")

    for i in range(len(students)):
        for j in range(i + 1, len(students)):

            name1 = students[i][0]
            average1 = students[i][3]

            name2 = students[j][0]
            average2 = students[j][3]

            if average1 > average2:
                print(name1, "performed better than", name2)

            elif average2 > average1:
                print(name2, "performed better than", name1)

            else:
                print(name1, "and", name2,
                      "have equal averages")
