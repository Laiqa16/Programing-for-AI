import numpy as np
import time

n = 1_000_000

numbers_list = list(range(n))
numbers_arr = np.arange(n)

# Measure list execution time
start = time.perf_counter()
square_list = [x ** 2 for x in numbers_list]
list_time = time.perf_counter() - start

# Measure NumPy array execution time
start = time.perf_counter()
squares_arr = numbers_arr ** 2
arr_time = time.perf_counter() - start

# Display results in milliseconds
print(f"List loop    : {list_time * 1000:,.2f} ms")
print(f"NumPy array  : {arr_time * 1000:,.2f} ms")

# Calculate speed comparison
print(f"NumPy is about: {list_time / arr_time:,.2f}x faster")
